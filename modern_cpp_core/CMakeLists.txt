cmake_minimum_required(VERSION 3.16)
project(ModernCppCore VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_TSAN "Enable ThreadSanitizer" OFF)
option(ENABLE_ASAN "Enable AddressSanitizer" OFF)

if(ENABLE_TSAN)
	message(STATUS "ThreadSanitizer is ENABLED globally")
	add_compile_options(-fsanitize=thread -fPIE -pie -g -O1)
	add_link_options(-fsanitize=thread -pie)
elseif(ENABLE_ASAN)
	message(STATUS "AddressSanitizer is ENABLED globally")
	add_compile_options(-fsanitize=address -fno-omit-frame-pointer -g -O1)
	add_link_options(-fsanitize=address)
endif()

add_subdirectory(concurrency)
add_subdirectory(coroutines)
add_subdirectory(data_structures)
add_subdirectory(memory)
add_subdirectory(patterns)
add_subdirectory(utils)
enable_testing()
add_subdirectory(tests)
